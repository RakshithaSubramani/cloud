## Server (`server.py`)
import Pyro4

@Pyro4.expose
class Shop:
    def calculate_bill(self, name, qty):
        products = {
            "Laptop": 50000,
            "Mouse": 500,
            "Keyboard": 1000
        }

        if name not in products or qty <= 0:
            return "Invalid product or quantity"

        total = products[name] * qty

        if total >= 1000:
            discount = total * 0.10
        else:
            discount = 0

        final = total - discount

        return {
            "Product": name,
            "Quantity": qty,
            "Total": total,
            "Discount": discount,
            "Final Amount": final
        }

Relay_server.py


import socket
import threading
import json
import sys


N = 3
MODE = sys.argv[1] if len(sys.argv) > 1 else "ricart"


clients = {}
send_locks = {}
lock = threading.Lock()
ready = False




def send(pid, message):
    with lock:
        conn = clients.get(pid)
        slock = send_locks.get(pid)


    if conn:
        try:
            with slock:
                conn.sendall((json.dumps(message) + "\n").encode())
        except OSError:
            pass




def handle(conn):
    global ready


    reader = conn.makefile("r")
    pid = None


    try:
        registration = json.loads(reader.readline())
        pid = registration["pid"]


        with lock:
            clients[pid] = conn
            send_locks[pid] = threading.Lock()
            print("Connected:", pid)


            if len(clients) == N and not ready:
                ready = True
                all_ready = list(clients.keys())
            else:
                all_ready = []


        for p in all_ready:
            send(p, {"type": "READY"})


        if all_ready and MODE == "token":
            send(1, {"type": "TOKEN"})


        for line in reader:
            message = json.loads(line)
            target = message["to"]
            payload = message["payload"]
            send(target, payload)


    except (OSError, ValueError, KeyError):
        pass
    finally:
        if pid is not None:
            with lock:
                clients.pop(pid, None)
                send_locks.pop(pid, None)
            print("Disconnected:", pid)
        conn.close()




server = socket.socket()
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("127.0.0.1", 5000))
server.listen()


print("Relay server started:", MODE)


while True:
    conn, _ = server.accept()
    threading.Thread(
        target=handle, args=(conn,), daemon=True
    ).start()

import socket
import threading
import json
import sys
import time


pid = int(sys.argv[1])
N = 3


sock = socket.socket()
sock.connect(("127.0.0.1", 5000))
sock.sendall((json.dumps({"pid": pid}) + "\n").encode())


send_lock = threading.Lock()
condition = threading.Condition()
ready = threading.Event()


clock = 0
state = "RELEASED"
request_time = 0
replies = set()
deferred = []




def send(to, payload):
    message = {"to": to, "payload": payload}
    with send_lock:
        sock.sendall((json.dumps(message) + "\n").encode())




def receive_messages():
    global clock, state


    reader = sock.makefile("r")


    for line in reader:
        msg = json.loads(line)
        kind = msg["type"]


        if kind == "READY":
            ready.set()
            continue


        if kind == "REPLY":
            with condition:
                replies.add(msg["from"])
                condition.notify_all()


        elif kind == "REQUEST":
            sender = msg["from"]
            timestamp = msg["ts"]


            with condition:
                clock = max(clock, timestamp) + 1


                defer = (
                    state == "HELD"
                    or (
                        state == "WANTED"
                        and (request_time, pid) < (timestamp, sender)
                    )
                )


                if defer:
                    deferred.append(sender)


            if defer:
                print(f"P{pid} defers reply to P{sender}")
            else:
                send(sender, {"type": "REPLY", "from": pid})




threading.Thread(target=receive_messages, daemon=True).start()


print(f"P{pid} connected; waiting for other processes...")
ready.wait()
print(f"P{pid} ready. Type r to request CS, q to quit.")


while True:
    command = input().strip().lower()


    if command == "q":
        break


    if command != "r":
        continue


    with condition:
        if state != "RELEASED":
            print("Already requesting or using CS")
            continue


        clock += 1
        request_time = clock
        state = "WANTED"
        replies.clear()


    print(f"P{pid} sends REQUEST")
    for other in range(1, N + 1):
        if other != pid:
            send(other, {
                "type": "REQUEST",
                "from": pid,
                "ts": request_time
            })


    with condition:
        while len(replies) < N - 1:
            condition.wait()
        state = "HELD"


    print(f"P{pid} ENTERS CS")
    time.sleep(2)
    print(f"P{pid} EXITS CS")


    with condition:
        state = "RELEASED"
        pending = deferred[:]
        deferred.clear()


    for other in pending:
        send(other, {"type": "REPLY", "from": pid})


sock.close()

import socket
import threading
import json


clients = {}
queue = []
busy = False
lock = threading.Lock()




def send(conn, message, send_lock):
    with send_lock:
        conn.sendall((json.dumps(message) + "\n").encode())




def grant_next():
    global busy


    if queue and not busy:
        pid = queue[0]
        busy = True
        conn, send_lock = clients[pid]
        print("Coordinator sends GRANT to", pid)
        send(conn, {"type": "GRANT"}, send_lock)




def handle(conn):
    global busy


    reader = conn.makefile("r")
    pid = None
    send_lock = threading.Lock()


    try:
        registration = json.loads(reader.readline())
        pid = registration["pid"]


        with lock:
            clients[pid] = (conn, send_lock)


        print("Process connected:", pid)


        for line in reader:
            msg = json.loads(line)


            with lock:
                if msg["type"] == "REQUEST":
                    print(pid, "sends REQUEST")


                    if pid not in queue:
                        queue.append(pid)


                    grant_next()


                elif msg["type"] == "RELEASE":
                    print(pid, "sends RELEASE")


                    if queue and queue[0] == pid:
                        queue.pop(0)
                        busy = False
                        grant_next()


    except (OSError, ValueError, KeyError):
        pass
    finally:
        with lock:
            clients.pop(pid, None)
            if pid in queue:
                was_first = queue[0] == pid
                queue.remove(pid)


                if was_first:
                    busy = False
                    grant_next()


        conn.close()
        print("Disconnected:", pid)




server = socket.socket()
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("127.0.0.1", 5001))
server.listen()


print("Coordinator server started on port 5001")


while True:
    conn, _ = server.accept()
    threading.Thread(
        target=handle, args=(conn,), daemon=True
    ).start()

class Process:


    def __init__(self, name):
        self.name = name
        self.clock = 0


    def event(self):
        self.clock += 1
        print(self.name, "event:", self.clock)


    def send(self):
        self.clock += 1
        print(self.name, "sends message:", self.clock)
        return self.clock


    def receive(self, timestamp):
        self.clock = max(self.clock, timestamp) + 1
        print(self.name, "receives message:", self.clock)

Vector clock:


class Process:


    def __init__(self, pid, n):
        self.pid = pid
        self.clock = [0] * n


    def event(self):
        self.clock[self.pid] += 1
        print("P", self.pid + 1, "event:", self.clock)


    def send(self):
        self.clock[self.pid] += 1
        print("P", self.pid + 1, "sends:", self.clock)
        return self.clock.copy()


    def receive(self, received):
        for i in range(len(self.clock)):
            self.clock[i] = max(self.clock[i], received[i])


        self.clock[self.pid] += 1


        print("P", self.pid + 1, "receives:", self.clock)




p1 = Process(0, 3)
p2 = Process(1, 3)


p1.event()


message = p1.send()


p2.receive(message)


p2.event()

6. Server consolidation — check resource usage
top
free -h
df -h
* top → CPU and running processes. Press q to exit.
* free -h → memory (RAM) usage.
* df -h → disk space usage.
The idea is to move services or files to a suitable VM, verify they work, and then shut down unused VMs.
7. Fault tolerance — backup and recovery
Install the required tools on the primary and backup VMs:
sudo apt update
sudo apt install rsync openssh-server -y
sudo systemctl enable --now ssh
On the backup VM:
mkdir -p ~/backup
On the primary VM, replace USERNAME and BACKUP_VM_IP with the correct values:
rsync -av /var/www/html/ USERNAME@BACKUP_VM_IP:/home/USERNAME/backup/
On the backup VM, verify:
ls -l ~/backup
Also remember the VMware options:
* Full Clone → creates a separate copy of a VM.
* Snapshot → saves a VM's state so you can return to it later.
Exercise 6 — Task Scheduling & Load Balancing
1. Create the script
Open Terminal on Ubuntu:
nano script.sh
Type:
#!/bin/bash
echo "Hello Selcia"
Save: Ctrl + O → Enter → Ctrl + X.
2. Make the script executable and test it
chmod +x script.sh
./script.sh
Expected output:
Hello Selcia
3. Schedule the script using cron
Open the cron editor:
crontab -e
If asked to choose an editor, select nano.
Add this line at the bottom. Replace USERNAME with your actual Ubuntu username:
* * * * * /bin/bash /home/USERNAME/script.sh >> /home/USERNAME/output.log 2>&1
Save and exit: Ctrl + O → Enter → Ctrl + X.
What does * * * * * mean? Run the script every minute.
4. Check the scheduled output
Wait one or two minutes, then run:
cat output.log
To watch new output appear:
tail -f output.log
Press Ctrl + C to stop watching.
To check cron activity:
grep CRON /var/log/syslog | tail
5. Load balancing in Exercise 6
Use the same HAProxy procedure from Exercise 5:
1. Start all web server VMs.
2. Make sure Apache is running on each web server.
3. Find each web server's IP using hostname -I.
4. Install HAProxy on the load balancer VM.
5. Configure the frontend, backend and roundrobin algorithm.
6. Validate the configuration and start HAProxy.
7. Test using curl http://LOAD_BALANCER_IP.
6. Optional advanced scheduling — nice
nice -n 10 ./script.sh
This runs the program with a lower CPU scheduling priority than a normal-priority process in typical Linux configurations. It does not schedule the program to run at a particular time.
